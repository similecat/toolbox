2. Build Any New Worker Service & Call Peer Workers Directly (Zero Central Server Hop)
from worker_sdk import ToolboxWorkerRuntime

runtime = ToolboxWorkerRuntime(
    server_url="{{ server_base_url }}",
    api_key="{{ key_placeholder }}",
    service_type="podcast-pipeline",
    service_version="1.0.0",
    max_concurrency=4,
    capabilities={"formats": ["wav", "mp3"]},
)

@runtime.operation("narrate_summary")
def narrate_summary(payload, ctx):
    # Direct Peer-to-Peer call to an online 'tts' worker — audio bytes travel Worker -> Worker!
    wav_resp = runtime.call_peer_raw("tts", "synthesize", {
        "text": payload["text"],
        "language": payload.get("language", "en"),
    })
    return {"audio_bytes": len(wav_resp.body), "tts_worker": wav_resp.headers.get("x-worker-id")}

with runtime:
    runtime.serve_forever()
